Esta página abriga a totalidade dos scripts SQL que compõem o pipeline de dados do Monitor Gestante. O código aqui apresentado é a versão exata utilizada em produção no Google BigQuery.

📖 Como ler esta documentação (Para todos):

Pense neste monitor como um sistema de "Vigilância de Saúde 24h". Ele lê milhares de dados todos os dias para responder perguntas simples: "Quem está grávida?", "Quem precisa de ajuda urgente?", "Quem abandonou o tratamento?". Abaixo, explicamos como cada peça dessa engrenagem funciona.

1. Identificação de Gestações (proced_1_gestacoes.sql)

Objetivo

Identificar e consolidar gestações únicas, definindo início, fim, fase atual e equipe responsável.

🔍 O Detetive de Inícios (Explicação Simplificada):

Imagine que o sistema é um detetive procurando pistas nos prontuários. A principal "pista" é o código Z32.1 (Confirmação de Gravidez). Mas o detetive é esperto: se ele encontrar uma nova confirmação apenas 10 dias depois da primeira, ele sabe que é a mesma gravidez. Ele só considera uma nova história se passarem mais de 60 dias sem notícias.

Lógica Detalhada (CTEs)

  • cadastro_paciente: Extrai dados básicos e calcula idade.
  • eventos_brutos/inicios/finais: Filtra CIDs Z32.1, Z34.x, Z35.x (Ativos/Resolvidos).
  • Lógica de Agrupamento: Considera nova gestação após intervalo de 60 dias.
  • Calculo de Datas: DPP (40 semanas) e Fim Estimado (299 dias).

Principais Regras de Negócio

  • Janela de 60 dias: Intervalo mínimo para considerar um novo ciclo.
  • Puerpério: Definido como 45 dias após o encerramento.

Código SQL Integral

-- CREATE OR REPLACE PROCEDURE `rj-sms-sandbox.sub_pav_us.proced_1_gestacoes`()
-- BEGIN
CREATE OR REPLACE TABLE `rj-sms-sandbox.sub_pav_us._gestacoes` AS
WITH
    cadastro_paciente AS (
        SELECT id_paciente, nome, DATE_DIFF(CURRENT_DATE(), data_nascimento, YEAR) AS idade_gestante
        FROM `rj-sms.saude_historico_clinico.paciente`        
    ),
    eventos_brutos AS (
        SELECT id_hci, paciente.id_paciente AS id_paciente, paciente_cpf as cpf, cp.nome, cp.idade_gestante,
            c.id AS cid, c.situacao AS situacao_cid, SAFE.PARSE_DATE('%Y-%m-%d', SUBSTR(c.data_diagnostico, 1, 10)) AS data_evento,
            CASE WHEN c.id = 'Z321' OR c.id LIKE 'Z34%' OR c.id LIKE 'Z34%' OR c.id LIKE 'Z35%' THEN 'gestacao' ELSE NULL END AS tipo_evento
        FROM `rj-sms.saude_historico_clinico.episodio_assistencial`
            LEFT JOIN UNNEST (condicoes) c 
            INNER JOIN cadastro_paciente cp ON paciente.id_paciente = cp.id_paciente
        WHERE c.data_diagnostico IS NOT NULL AND c.data_diagnostico != '' AND c.situacao IN ('ATIVO', 'RESOLVIDO')
            AND (c.id = 'Z321' OR c.id LIKE 'Z34%' OR c.id LIKE 'Z35%') AND paciente.id_paciente IS NOT NULL
    ),
    inicios_brutos AS ( SELECT * FROM eventos_brutos WHERE tipo_evento = 'gestacao' AND situacao_cid = 'ATIVO' ),
    finais AS ( SELECT * FROM eventos_brutos WHERE tipo_evento = 'gestacao' AND situacao_cid = 'RESOLVIDO' ),
    inicios_com_grupo AS (
        SELECT *, CASE WHEN LAG(data_evento) OVER (PARTITION BY id_paciente ORDER BY data_evento) IS NULL THEN 1
            WHEN DATE_DIFF(data_evento, LAG(data_evento) OVER(PARTITION BY id_paciente ORDER BY data_evento), DAY) >= 60 THEN 1
            ELSE 0 END AS nova_ocorrencia_flag
        FROM inicios_brutos
    ),
    grupos_inicios AS ( SELECT *, SUM(nova_ocorrencia_flag) OVER(PARTITION BY id_paciente ORDER BY data_evento) AS grupo_id FROM inicios_com_grupo ),
    inicios_deduplicados AS ( SELECT * FROM (SELECT *, ROW_NUMBER() OVER(PARTITION BY id_paciente, grupo_id ORDER BY data_evento DESC) AS rn FROM grupos_inicios) WHERE rn = 1 ),
    gestacoes_unicas AS (
        SELECT i.id_hci, i.id_paciente, i.cpf, i.nome, i.idade_gestante, i.data_evento AS data_inicio,
            (SELECT MIN(f.data_evento) FROM finais f WHERE f.id_paciente = i.id_paciente AND f.data_evento > i.data_evento) AS data_fim,
            ROW_NUMBER() OVER(PARTITION BY i.id_paciente ORDER BY i.data_evento) AS numero_gestacao,
            CONCAT(i.id_paciente, '-', CAST(ROW_NUMBER() OVER(PARTITION BY i.id_paciente ORDER BY i.data_evento) AS STRING)) AS id_gestacao
        FROM inicios_deduplicados i
    ),
    gestacoes_com_status AS (
        SELECT *, CASE WHEN data_fim IS NOT NULL THEN data_fim WHEN DATE_ADD(data_inicio, INTERVAL 299 DAY) <= CURRENT_DATE() THEN DATE_ADD(data_inicio, INTERVAL 299 DAY) ELSE NULL END AS data_fim_efetiva,
            DATE_ADD(data_inicio, INTERVAL 40 WEEK) AS dpp FROM gestacoes_unicas
    ),
    filtrado AS (
        SELECT gcs.*, CASE WHEN gcs.data_fim IS NULL AND DATE_ADD(gcs.data_inicio, INTERVAL 299 DAY) > CURRENT_DATE() THEN 'Gestação' WHEN gcs.data_fim IS NOT NULL AND DATE_DIFF(CURRENT_DATE(), gcs.data_fim, DAY) <= 45 THEN 'Puerpério' ELSE 'Encerrada' END AS fase_atual,
            CASE WHEN DATE_DIFF(CURRENT_DATE(), gcs.data_inicio, WEEK) <= 13 THEN '1º trimestre' WHEN DATE_DIFF(CURRENT_DATE(), gcs.data_inicio, WEEK) BETWEEN 14 AND 27  THEN '2º trimestre' WHEN DATE_DIFF(CURRENT_DATE(), gcs.data_inicio, WEEK) >= 28 THEN '3º trimestre' ELSE 'Data inválida ou encerrada' END AS trimestre_atual_gestacao
        FROM gestacoes_com_status gcs
    ),
    unnested_equipes AS ( SELECT p.dados.id_paciente AS id_paciente, eq.datahora_ultima_atualizacao, eq.nome AS equipe_nome, eq.clinica_familia.nome AS clinica_nome FROM `rj-sms.saude_historico_clinico.paciente` p left join UNNEST (p.equipe_saude_familia) AS eq ),
    equipe_durante_gestacao AS ( SELECT f.id_gestacao, eq.equipe_nome, eq.clinica_nome, ROW_NUMBER() OVER(PARTITION BY f.id_gestacao ORDER BY eq.datahora_ultima_atualizacao DESC) AS rn FROM filtrado f LEFT JOIN unnested_equipes eq ON f.id_paciente = eq.id_paciente AND DATE(eq.datahora_ultima_atualizacao) <= COALESCE(f.data_fim_efetiva, CURRENT_DATE()) ),
    equipe_durante_final AS ( SELECT id_gestacao, equipe_nome, clinica_nome FROM equipe_durante_gestacao WHERE rn = 1 )
SELECT filtrado.*, edf.equipe_nome, edf.clinica_nome FROM filtrado LEFT JOIN equipe_durante_final edf ON filtrado.id_gestacao = edf.id_gestacao;
-- END;

2. Atendimento Pré-Natal APS (proced_2_atd_prenatal_aps.sql)

Objetivo

Consolidar atendimentos de pré-natal na APS, incluindo antropometria, sinais vitais e prescrições.

📒 O Diário da Gestante Digital:

Aqui transformamos o prontuário em um diário organizado. O sistema varre as consultas para responder: "Ela está ganhando peso direitinho?" (cruzando peso x altura) e "Ela foi hoje na consulta?". Ele ignora consultas que não sejam de pré-natal para não "sujar" o diário.

Lógica Detalhada (CTEs)

  • Peso Inicial: Busca o peso mais próximo da DUM (prioridade anterior, fallback posterior).
  • Altura Moda: Calcula a altura mais frequente para estabilidade dos índices.
  • IMC de Início: Classifica o estado nutricional inicial.
  • Filtro SOAP: Identifica consultas de médicos e enfermeiros da ESF via Vitacare.
  • Ganho de Peso: Calcula a evolução ponderal ao longo das consultas.

Principais Regras de Negócio

  • Classificação IMC: Segue faixas padrão (Baixo peso, Eutrófico, Sobrepeso, Obesidade).
  • IG da Consulta: Calculada em semanas para permitir análise por trimestre.

Código SQL Integral

CREATE OR REPLACE PROCEDURE `rj-sms-sandbox.sub_pav_us.proced_2_atd_prenatal_aps`()
BEGIN
CREATE OR REPLACE TABLE `rj-sms-sandbox.sub_pav_us._atendimentos_prenatal_aps` AS
WITH marcadores_temporais AS (
 SELECT id_gestacao, id_paciente, cpf, nome, numero_gestacao, idade_gestante, data_inicio, data_fim, data_fim_efetiva, fase_atual
 FROM `rj-sms-sandbox.sub_pav_us._gestacoes`
),
peso_anterior_dum AS (
 SELECT mt.id_gestacao, mt.id_paciente, ea.entrada_data, ea.medidas.peso, ROW_NUMBER() OVER (PARTITION BY mt.id_gestacao ORDER BY ea.entrada_data DESC) AS rn
 FROM `rj-sms.saude_historico_clinico.episodio_assistencial` ea
 JOIN marcadores_temporais mt ON ea.paciente.id_paciente = mt.id_paciente
 WHERE ea.medidas.peso IS NOT NULL AND ea.entrada_data < mt.data_inicio AND ea.entrada_data >= DATE_SUB(mt.data_inicio, INTERVAL 180 DAY)
),
peso_posterior_dum AS (
 SELECT mt.id_gestacao, mt.id_paciente, ea.entrada_data, ea.medidas.peso, ROW_NUMBER() OVER (PARTITION BY mt.id_gestacao ORDER BY ea.entrada_data ASC) AS rn
 FROM `rj-sms.saude_historico_clinico.episodio_assistencial` ea
 JOIN marcadores_temporais mt ON ea.paciente.id_paciente = mt.id_paciente
 WHERE ea.medidas.peso IS NOT NULL AND ea.entrada_data >= mt.data_inicio
),
peso_proximo_inicio AS (
  SELECT id_gestacao, id_paciente, entrada_data, peso, 'ANTERIOR_DUM' AS origem_peso FROM peso_anterior_dum WHERE rn = 1
  UNION ALL
  SELECT id_gestacao, id_paciente, entrada_data, peso, 'POSTERIOR_DUM' AS origem_peso FROM peso_posterior_dum WHERE rn = 1 AND id_gestacao NOT IN (SELECT id_gestacao FROM peso_anterior_dum WHERE rn = 1)
),
alturas_filtradas AS (
 SELECT mt.id_gestacao, ea.paciente.id_paciente, ea.medidas.altura, DATE_DIFF(mt.data_inicio, ea.entrada_data, DAY) AS dias_antes_inicio, DATE_DIFF(ea.entrada_data, COALESCE(mt.data_fim_efetiva, CURRENT_DATE()), DAY) AS dias_apos_inicio
 FROM `rj-sms.saude_historico_clinico.episodio_assistencial` ea
 JOIN marcadores_temporais mt ON ea.paciente.id_paciente = mt.id_paciente
 WHERE ea.medidas.altura IS NOT NULL
),
altura_preferencial AS (
 SELECT id_gestacao, id_paciente, CAST(altura AS FLOAT64) AS altura_cm, COUNT(*) AS freq, ROW_NUMBER() OVER (PARTITION BY id_gestacao ORDER BY COUNT(*) DESC) AS rn
 FROM alturas_filtradas WHERE dias_antes_inicio <= 365 AND dias_apos_inicio <= 0 GROUP BY id_gestacao, id_paciente, altura
),
altura_fallback AS (
 SELECT id_gestacao, id_paciente, CAST(altura AS FLOAT64) AS altura_cm, COUNT(*) AS freq, ROW_NUMBER() OVER (PARTITION BY id_gestacao ORDER BY COUNT(*) DESC) AS rn
 FROM alturas_filtradas GROUP BY id_gestacao, id_paciente, altura
),
altura_moda_completa AS (
 SELECT * FROM altura_preferencial WHERE rn = 1 UNION ALL SELECT * FROM altura_fallback WHERE id_gestacao NOT IN (SELECT id_gestacao FROM altura_preferencial WHERE rn = 1)
),
peso_altura_inicio AS (
 SELECT p.id_gestacao, p.id_paciente, p.peso, p.entrada_data AS data_peso_inicio, a.altura_cm / 100 AS altura_m, ROUND(p.peso / POW(a.altura_cm / 100, 2), 1) AS imc_inicio,
   CASE WHEN ROUND(p.peso / POW(a.altura_cm / 100, 2), 1) < 18 THEN 'Baixo peso' WHEN ROUND(p.peso / POW(a.altura_cm / 100, 2), 1) < 25 THEN 'Eutrófico' WHEN ROUND(p.peso / POW(a.altura_cm / 100, 2), 1) < 30 THEN 'Sobrepeso' ELSE 'Obesidade' END AS classificacao_imc_inicio
 FROM peso_proximo_inicio p JOIN altura_moda_completa a ON p.id_gestacao = a.id_gestacao
),
atendimentos_filtrados AS (
 SELECT ea.id_hci, ea.paciente.id_paciente, ea.entrada_data, ea.estabelecimento.nome AS estabelecimento, ea.profissional_saude_responsavel.nome AS profissional_nome, ea.profissional_saude_responsavel.especialidade AS profissional_categoria,
  ANY_VALUE(ea.medidas.altura) AS altura, ANY_VALUE(ea.medidas.peso) AS peso, ANY_VALUE(ea.medidas.imc) AS imc, ANY_VALUE(ea.medidas.pressao_sistolica) AS pressao_sistolica, ANY_VALUE(ea.medidas.pressao_diastolica) AS pressao_diastolica,
  ANY_VALUE(ea.motivo_atendimento) AS motivo_atendimento, ANY_VALUE(ea.desfecho_atendimento) AS desfecho_atendimento, STRING_AGG(c.id, ', ' ORDER BY c.id) AS cid_string
 FROM `rj-sms.saude_historico_clinico.episodio_assistencial` ea LEFT JOIN UNNEST(ea.condicoes) AS c
 WHERE ea.subtipo = 'Atendimento SOAP' AND LOWER(ea.prontuario.fornecedor) = 'vitacare'
 AND ea.profissional_saude_responsavel.especialidade IN ('Médico da estratégia de saúde da família','Enfermeiro da estratégia saúde da família','Enfermeiro - Modelo B','Médico Clínico','Médico Ginecologista e Obstetra - NASF','Médico Ginecologista - Modelo B','Médico Clinico - Modelo B','Enfermeiro obstétrico','Enfermeiro','Enfermeiro Obstetrico - Nasf','Médico Generalista','Médico de Família e Comunidade','Farmacêutico Hospitalar e Clinico - NASF')
 GROUP BY 1,2,3,4,5,6,7
),
atendimentos_gestacao AS (
 SELECT af.*, mt.id_gestacao, mt.data_inicio, mt.data_fim_efetiva, mt.fase_atual, DATE_DIFF(af.entrada_data, mt.data_inicio, WEEK) AS ig_consulta,
   CASE WHEN DATE_DIFF(af.entrada_data, mt.data_inicio, WEEK) <= 13 THEN 1 WHEN DATE_DIFF(af.entrada_data, mt.data_inicio, WEEK) <= 27 THEN 2 ELSE 3 END AS trimestre_consulta
 FROM atendimentos_filtrados af JOIN marcadores_temporais mt ON af.id_paciente = mt.id_paciente AND af.entrada_data BETWEEN mt.data_inicio AND COALESCE(mt.data_fim_efetiva, CURRENT_DATE())
),
prescricoes_aggregadas AS (
 SELECT ea.id_hci, STRING_AGG(p.nome, ', ') AS prescricoes
 FROM `rj-sms.saude_historico_clinico.episodio_assistencial` ea LEFT join UNNEST(ea.prescricoes) AS p
 WHERE ea.subtipo = 'Atendimento SOAP' AND LOWER(ea.prontuario.fornecedor) = 'vitacare' GROUP BY ea.id_hci
),
consultas_enriquecidas AS (
 SELECT ag.*, presc.prescricoes, ROW_NUMBER() OVER (PARTITION BY ag.id_gestacao ORDER BY ag.entrada_data) AS numero_consulta, pai.peso AS peso_inicio, pai.data_peso_inicio, pai.altura_m, pai.imc_inicio, pai.classificacao_imc_inicio,
   ag.peso - pai.peso AS ganho_peso_acumulado, ROUND(ag.peso / POW(pai.altura_m, 2), 1) AS imc_consulta
 FROM atendimentos_gestacao ag LEFT JOIN prescricoes_aggregadas presc ON ag.id_hci = presc.id_hci LEFT JOIN peso_altura_inicio pai ON ag.id_gestacao = pai.id_gestacao
)
SELECT id_gestacao, id_paciente, entrada_data AS data_consulta, numero_consulta, ig_consulta, trimestre_consulta, fase_atual, peso_inicio, data_peso_inicio, altura_m AS altura_inicio, imc_inicio, classificacao_imc_inicio, peso, imc_consulta, ganho_peso_acumulado, pressao_sistolica, pressao_diastolica, motivo_atendimento AS descricao_s, cid_string, desfecho_atendimento AS desfecho, prescricoes, estabelecimento, profissional_nome, profissional_categoria
FROM consultas_enriquecidas WHERE fase_atual = 'Gestação' ORDER BY data_consulta DESC;
END;

3. Visitas ACS (proced_3_visitas_acs_gestacao.sql)

Objetivo

Acompanhar a frequência e autoria das visitas domiciliares realizadas pelos ACS.

🏠 Olheiros da Comunidade:

Muitas vezes a saúde vai até a casa da pessoa. Este script conta quantas vezes o Agente Comunitário (ACS) bateu na porta da gestante. É vital para saber quem está sendo acompanhado de perto, mesmo sem ir ao posto.

Lógica Detalhada (CTEs)

  • Filtro Profissional: Seleciona estritamente 'Agente comunitário de saúde'.
  • Subtipo: Considera apenas 'Visita Domiciliar' do Vitacare.
  • Vínculo Temporal: Associa a visita à gestação se ocorrer durante o ciclo.

Principais Regras de Negócio

  • Numeração Sequencial: Identifica a 1ª, 2ª, nth visita por gestação.

Código SQL Integral

CREATE OR REPLACE PROCEDURE `rj-sms-sandbox.sub_pav_us.proced_3_visitas_acs_gestacao`()
BEGIN
CREATE OR REPLACE TABLE `rj-sms-sandbox.sub_pav_us._visitas_acs_gestacao` AS
WITH marcadores_temporais AS (
 SELECT id_gestacao, id_paciente, cpf, nome AS nome_gestante, numero_gestacao, idade_gestante, data_inicio, data_fim, data_fim_efetiva, clinica_nome AS unidade_APS_PN, equipe_nome AS equipe_PN_APS
 FROM `rj-sms-sandbox.sub_pav_us._gestacoes`
),
visitas_com_join AS (
 SELECT ea.id_hci, ea.paciente.id_paciente, ea.entrada_data, ea.estabelecimento.nome AS nome_estabelecimento, ea.profissional_saude_responsavel.nome AS nome_profissional, mt.id_gestacao, mt.cpf, mt.nome_gestante, mt.numero_gestacao, mt.idade_gestante, mt.data_inicio, mt.data_fim, mt.data_fim_efetiva
 FROM `rj-sms.saude_historico_clinico.episodio_assistencial` ea
 JOIN marcadores_temporais mt ON ea.paciente.id_paciente = mt.id_paciente AND ea.entrada_data BETWEEN mt.data_inicio AND coalesce(mt.data_fim_efetiva, CURRENT_DATE())
 WHERE ea.prontuario.fornecedor = 'vitacare' AND ea.profissional_saude_responsavel.especialidade = 'Agente comunitário de saúde' AND ea.subtipo = 'Visita Domiciliar'
)
SELECT *, ROW_NUMBER() OVER (PARTITION BY id_gestacao ORDER BY entrada_data) AS numero_visita
FROM visitas_com_join ORDER BY id_gestacao, entrada_data;
END;

4. Consultas Emergenciais (proced_4_consultas_emergenciais.sql)

Objetivo

Monitorar intercorrências através de atendimentos na rede de emergência (Vitai).

🚨 O Alerta Vermelho:

Se uma gestante passa mal e vai para a UPA ou Hospital, a Clínica da Família precisa saber. Este "robô" fica vigiando o sistema de hospitais (Vitai). Se ele vê o nome de uma gestante lá, ele avisa imediatamente: "Atenção, sua paciente teve uma emergência!".

Lógica Detalhada (CTEs)

  • Agrupamento de CIDs: Consolida múltiplos diagnósticos do mesmo atendimento em strings.
  • Origem Vitai: Filtra atendimentos hospitalares/UPA.
  • Mapeamento IG: Calcula a idade gestacional exata no momento da emergência.

Principais Regras de Negócio

  • Alerta de Intercorrência: Permite à APS saber quando uma gestante buscou a rede hospitalar.

Código SQL Integral

CREATE OR REPLACE PROCEDURE `rj-sms-sandbox.sub_pav_us.proced_4_consultas_emergenciais`()
BEGIN
CREATE OR REPLACE TABLE `rj-sms-sandbox.sub_pav_us._consultas_emergenciais` AS
WITH marcadores_temporais AS (
 SELECT id_gestacao, id_paciente, cpf, nome AS nome_gestante, numero_gestacao, idade_gestante, data_inicio, data_fim, data_fim_efetiva, fase_atual, clinica_nome AS unidade_APS_PN, equipe_nome AS equipe_PN_APS
 FROM `rj-sms-sandbox.sub_pav_us._gestacoes`
),
cids_agrupados AS (
 SELECT ea.id_hci, STRING_AGG(DISTINCT c.id ORDER BY c.id) AS cids_emergencia, STRING_AGG(DISTINCT c.descricao ORDER BY c.descricao) AS descricoes_cids_emergencia
 FROM `rj-sms.saude_historico_clinico.episodio_assistencial` ea left join UNNEST(ea.condicoes) AS c
 WHERE ea.prontuario.fornecedor = 'vitai' AND ea.subtipo = 'Emergência' AND c.id IS NOT NULL GROUP BY ea.id_hci
)
SELECT mt.id_gestacao, mt.numero_gestacao, mt.id_paciente, mt.cpf, mt.nome_gestante, mt.idade_gestante, mt.data_inicio, mt.data_fim, mt.data_fim_efetiva, mt.unidade_APS_PN, mt.equipe_PN_APS, mt.fase_atual, ea.id_hci, ea.entrada_data AS data_consulta, DATE_DIFF (ea.entrada_data, mt.data_inicio, WEEK) AS idade_gestacional_consulta, ROW_NUMBER() OVER (PARTITION BY mt.id_gestacao ORDER BY ea.entrada_data) AS numero_consulta, ea.motivo_atendimento, ea.desfecho_atendimento, cids.cids_emergencia, cids.descricoes_cids_emergencia, ea.profissional_saude_responsavel.nome AS nome_profissional, ea.profissional_saude_responsavel.especialidade AS especialidade_profissional, ea.estabelecimento.nome AS nome_estabelecimento
FROM `rj-sms.saude_historico_clinico.episodio_assistencial` ea JOIN marcadores_temporais mt ON ea.paciente.id_paciente = mt.id_paciente AND ea.entrada_data BETWEEN mt.data_inicio AND COALESCE(mt.data_fim_efetiva, CURRENT_DATE())
LEFT JOIN cids_agrupados cids ON ea.id_hci = cids.id_hci
WHERE ea.prontuario.fornecedor = 'vitai' AND ea.subtipo = 'Emergência' ORDER BY mt.id_gestacao, ea.entrada_data;
END;

5. Encaminhamentos de Alto Risco (proced_5_encaminhamentos.sql)

Objetivo

Identificar solicitações de regulação para Pré-Natal de Alto Risco (Pre-Natal AR).

🏥 O Pedido de Socorro (SISREG/SER):

Às vezes, a UBS (postinho) não consegue resolver sozinha. O médico precisa chamar um especialista. Este script verifica se esse "pedido de socorro" foi feito (no sistema SISREG ou SER) e se ele foi aceito. É crucial para gantir que casos graves não fiquem parados na fila.

Lógica Detalhada (CTEs)

  • Mapeamento Multissistema: Integra dados do SISREG (Municipal) e SER (Estadual).
  • Busca de IDs: Utiliza CPF e CNS para garantir o encontro dos registros.
  • Procedimentos Alvo: Filtra códigos específicos de risco gestacional.

Principais Regras de Negócio

  • Consolidação: Indica se houve encaminhamento, a origem (SISREG/SER) e o status atual.

Código SQL Integral

CREATE OR REPLACE PROCEDURE `rj-sms-sandbox.sub_pav_us.proced_5_encaminhamentos`()
BEGIN
CREATE OR REPLACE TABLE `rj-sms-sandbox.sub_pav_us._encaminhamentos_V2` AS
WITH gestacoes_definidas AS (
    SELECT id_gestacao, id_paciente, cpf, data_inicio, data_fim_efetiva FROM `rj-sms-sandbox.sub_pav_us._gestacoes` WHERE fase_atual = 'Gestação'
),
paciente_identificadores AS (
    SELECT g.id_paciente, p.cpf, cns_paciente FROM gestacoes_definidas g JOIN rj-sms.saude_historico_clinico.paciente p ON g.id_paciente = p.dados.id_paciente LEFT JOIN UNNEST(p.cns) AS cns_paciente QUALIFY ROW_NUMBER() OVER(PARTITION BY g.id_paciente, cns_paciente) = 1
),
sisreg_pre_filtrado AS (
    SELECT pi.id_paciente, pi.cpf, s.* FROM rj-sms.brutos_sisreg_api.solicitacoes s JOIN paciente_identificadores pi ON (s.paciente_cpf = pi.cpf AND pi.cpf IS NOT NULL AND pi.cpf != '') OR (s.paciente_cns = pi.cns_paciente AND pi.cns_paciente IS NOT NULL AND pi.cns_paciente != '') WHERE s.procedimento_id IN ('0703844','0703886','0737024','0710301','0710128')
),
ser_pre_filtrado AS (
    SELECT pi.id_paciente, pi.cpf, ser.* FROM `rj-sms-sandbox.sub_pav_us.BD SER PRÉ NATAL 2024_2025AGO` ser JOIN paciente_identificadores pi ON CAST(ser.cns AS STRING) = pi.cns_paciente AND pi.cns_paciente IS NOT NULL
),
encaminhamento_SISREG AS (
    SELECT g.id_gestacao, DATE(s.data_solicitacao) AS data_solicitacao, s.solicitacao_status, s.solicitacao_situacao, s.procedimento, s.procedimento_id, s.unidade_solicitante, s.medico_solicitante, s.operador_solicitante_nome, s.cid_id, ROW_NUMBER() OVER (PARTITION BY g.id_gestacao ORDER BY s.data_solicitacao ASC) as rn_solicitacao
    FROM gestacoes_definidas g JOIN sisreg_pre_filtrado s ON g.id_paciente = s.id_paciente WHERE DATE(s.data_solicitacao) BETWEEN g.data_inicio AND COALESCE(g.data_fim_efetiva, CURRENT_DATE())
),
encaminhamento_SER AS (
    SELECT g.id_gestacao, s.classificacao_risco, s.Recurso_Solicitado, s.Estado_Solicitacao, s.Codigo_cid, s.Descricao_cid, CAST(s.Dt_agendamento AS DATE) AS Dt_agendamento, CAST(s.Dt_execucao AS DATE) AS Dt_execucao, s.UnidadeExecutante, s.Unidade_Origem, ROW_NUMBER() OVER (PARTITION BY g.id_gestacao ORDER BY CAST(s.Dt_Solicitacao AS DATE) ASC) as rn_solicitacao
    FROM gestacoes_definidas g JOIN ser_pre_filtrado s ON g.id_paciente = s.id_paciente WHERE CAST(s.Dt_Solicitacao AS DATE) BETWEEN g.data_inicio AND COALESCE(g.data_fim_efetiva, CURRENT_DATE())
)
SELECT g.id_gestacao, g.id_paciente, g.cpf, CASE WHEN sis_sol.id_gestacao IS NOT NULL OR ser_sol.id_gestacao IS NOT NULL THEN 'Sim' ELSE 'Não' END AS houve_encaminhamento, CASE WHEN sis_sol.id_gestacao IS NOT NULL AND ser_sol.id_gestacao IS NOT NULL THEN 'Ambos' WHEN sis_sol.id_gestacao IS NOT NULL THEN 'SISREG' WHEN ser_sol.id_gestacao IS NOT NULL THEN 'SER' ELSE NULL END AS origem_encaminhamento, sis_sol.data_solicitacao AS sisreg_primeira_data_solicitacao, sis_sol.solicitacao_status AS sisreg_primeira_status, sis_sol.solicitacao_situacao AS sisreg_primeira_situacao, sis_sol.procedimento AS sisreg_primeira_procedimento_nome, sis_sol.procedimento_id AS sisreg_primeira_procedimento_id, sis_sol.cid_id AS sisreg_primeira_cid, sis_sol.unidade_solicitante AS sisreg_primeira_unidade_solicitante, sis_sol.medico_solicitante AS sisreg_primeira_medico_solicitante, ser_sol.classificacao_risco AS ser_classificacao_risco, ser_sol.Recurso_Solicitado AS ser_recurso_solicitado, ser_sol.Estado_Solicitacao AS ser_estado_solicitacao, ser_sol.Dt_agendamento AS ser_data_agendamento, ser_sol.Dt_execucao AS ser_data_execucao, ser_sol.UnidadeExecutante AS ser_unidade_executante, ser_sol.Codigo_cid AS ser_cid, ser_sol.Descricao_cid AS ser_descricao_cid, ser_sol.Unidade_Origem AS ser_unidade_origem
FROM gestacoes_definidas g LEFT JOIN (SELECT * FROM encaminhamento_SISREG WHERE rn_solicitacao = 1) sis_sol ON g.id_gestacao = sis_sol.id_gestacao LEFT JOIN (SELECT * FROM encaminhamento_SER WHERE rn_solicitacao = 1) ser_sol ON g.id_gestacao = ser_sol.id_gestacao;
END;

6. Linha do Tempo Master (proced_6_linha_tempo.sql)

Objetivo

O "Cérebro" do monitoramento. Consolida todos os dados em uma visão 360º da gestante.

🧠 O Cérebro de Tudo (Linha do Tempo):

Este é o script mais inteligente. Ele pega as peças soltas (a consulta de peso do script 2, o alerta de emergência do script 4, o remédio do script 8) e monta o quebra-cabeça. Além disso, ele diagnóstica riscos automaticamente: se a pressão da paciente deu alta 3 vezes, ele coloca um carimbo de "Hipertensa nela", mesmo que ninguém tenha anotado isso manualmente.

Detalhamento Técnico das CTEs

1. Base e Diagnósticos

filtrado: Seleciona o universo de gestações já identificadas no proced_1.

filtrado AS (
    SELECT *
    FROM `rj-sms-sandbox.sub_pav_us._gestacoes`
)

condicoes_gestantes_raw: Coleta todos os diagnósticos (CIDs) registrados no histórico clínico, normalizando a data.

condicoes_gestantes_raw AS (
    SELECT ea.paciente.id_paciente, c.id AS cid, SAFE.PARSE_DATE('%Y-%m-%d', SUBSTR(c.data_diagnostico, 1, 10)) AS data_diagnostico, c.situacao
    FROM `rj-sms.saude_historico_clinico.episodio_assistencial` ea
    LEFT JOIN UNNEST (condicoes) c
    WHERE c.situacao IN ('ATIVO', 'RESOLVIDO') AND c.id IS NOT NULL 
)

2. Governança Territorial (cad_e_atd)

Cruza dados de cadastro com o histórico real de atendimentos para definir a unidade "de fato". 

atendimentos_por_unidade AS (
    SELECT lt.id_paciente, COALESCE(NULLIF(UPPER(TRIM(apa.estabelecimento)), ''), 'SEM INFORMACAO') AS unidade_norm,
           COUNT(*) AS total_atendimentos, MAX(apa.data_consulta) AS ultima_data_atendimento
    FROM linha_tempo_base lt
    JOIN `rj-sms-sandbox.sub_pav_us._atendimentos_prenatal_aps` apa ON lt.id_gestacao = apa.id_gestacao
    GROUP BY lt.id_paciente, unidade_norm
)

Categorização Clínica (Flags de Risco)

O motor de regras clínicas transforma registros brutos em sinalizadores binários (0 ou 1).

1. Grupo Hipertensão (HAS)

Monitora três estados distintos para diferenciar a cronicidade da intercorrência gestacional:

  • Hipertensão Prévia: Identifica a condição crônica preexistente.
    • CIDs: I10-I15 ou O10
    MAX(CASE WHEN (REGEXP_CONTAINS(cg.cid, r'I1[0-5]') OR REGEXP_CONTAINS(cg.cid, r'O10')) 
    THEN 1 ELSE 0 END) AS hipertensao_previa
  • Pré-eclâmpsia: Identifica riscos agudos específicos do ciclo gravídico observados durante a gestação.
    • CIDs: O11 ou O14
    MAX(CASE WHEN (cg.cid LIKE '%O11%' OR cg.cid LIKE '%O14%') 
    THEN 1 ELSE 0 END) AS preeclampsia
  • Hipertensão Não Especificada: Captura registros de monitoramento gestacional sem diagnóstico fechado de pré-eclâmpsia.
    • CIDs: O16
    MAX(CASE WHEN cg.cid = '%O16%' THEN 1 ELSE 0 END) AS hipertensao_nao_especificada

2. Grupo Diabetes

Diferencia diabetes mellitus preexistente de diabetes gestacional (DMG):

  • Diabetes Prévio: Diagnosticado antes do início ou até o desfecho.
    • CIDs: E10-E14 ou O24.0 a O24.3
    MAX(CASE WHEN (REGEXP_CONTAINS(cg.cid, r'\bE1[0-4]\b') OR REGEXP_CONTAINS(cg.cid, r'\bO24[0-3]\b'))
    AND cg.data_diagnostico < COALESCE(f.data_fim_efetiva, f.dpp, CURRENT_DATE()) 
    THEN 1 ELSE 0 END) AS diabetes_previo
  • Diabetes Gestacional: Diagnosticado estritamente durante a gestação.
    • CIDs: O24.4
    MAX(CASE WHEN cg.cid = 'O244' AND cg.data_diagnostico BETWEEN f.data_inicio 
    AND COALESCE(f.data_fim_efetiva, f.dpp, CURRENT_DATE()) THEN 1 ELSE 0 END) AS diabetes_gestacional
  • Diabetes Não Especificado: Diagnosticado durante a gestação.
    • CIDs: O24.9
    MAX(CASE WHEN cg.cid = 'O249' AND cg.data_diagnostico BETWEEN f.data_inicio 
    AND COALESCE(f.data_fim_efetiva, f.dpp, CURRENT_DATE()) THEN 1 ELSE 0 END) AS diabetes_nao_especificado

3. Grupo Infecciosas

Monitoramento contínuo de patologias de transmissão vertical:

  • HIV: Busca diagnósticos B20-B24 ou marcador Z21. 
    MAX(CASE WHEN (REGEXP_CONTAINS(cg.cid, r'\bB2[0-4]\b') OR cg.cid = 'Z21') 
    AND cg.data_diagnostico <= COALESCE(f.data_fim_efetiva, f.dpp, CURRENT_DATE()) 
    THEN 1 ELSE 0 END) AS hiv
  • Sífilis: Janela estendida (30 dias pré-DUM) para capturar infecções periconcepcionais.
    • CIDs: A51-A53 (Primária, Secundária e Latente)
    MAX(CASE WHEN REGEXP_CONTAINS(cg.cid, r'\bA5[1-3]\b') AND cg.data_diagnostico BETWEEN 
    DATE_SUB(f.data_inicio, INTERVAL 30 DAY) AND COALESCE(f.data_fim_efetiva, f.dpp, CURRENT_DATE()) 
    THEN 1 ELSE 0 END) AS sifilis
  • Tuberculose: Janela de 6 meses pré-DUM.
    • CIDs: A15-A19
    MAX(CASE WHEN REGEXP_CONTAINS(cg.cid, r'\bA1[5-9]\b') AND cg.data_diagnostico BETWEEN 
    DATE_SUB(f.data_inicio, INTERVAL 6 MONTH) AND COALESCE(f.data_fim_efetiva, f.dpp, CURRENT_DATE()) 
    THEN 1 ELSE 0 END) AS tuberculose

4. Outros Riscos (Autoimunes e Reprodução)

  • Doenças Autoimunes: Lúpus (M32) e Síndrome Antifosfolipídio (D68.6/D686). 
    MAX(CASE WHEN (cg.cid LIKE 'M32%' OR cg.cid = 'D686' OR cg.cid = 'D68.6') 
    AND cg.data_diagnostico BETWEEN f.data_inicio AND COALESCE(f.data_fim_efetiva, f.dpp, CURRENT_DATE()) 
    THEN 1 ELSE 0 END) AS doenca_autoimune_cid
  • Reprodução Assistida: Marcadores de gestação via fertilização (Z31.2, Z31.3, Z31.8, Z31.9). 
    MAX(CASE WHEN cg.cid IN ('Z312','Z313','Z318','Z319') AND cg.data_diagnostico 
    BETWEEN f.data_inicio AND COALESCE(f.data_fim_efetiva, f.dpp, CURRENT_DATE()) 
    THEN 1 ELSE 0 END) AS reproducao_assistida_cid

7. Categorias de Risco BI (proced_7_categorias_risco.sql)

Objetivo

Desnormalizar os riscos para facilitar a criação de filtros em ferramentas de BI (Tableau/Power BI).

🏷️ O Organizador de Etiquetas:

Para facilitar a vida de quem faz gráficos e relatórios, este script pega a lista complexa de problemas de uma paciente e transforma em etiquetas simples: "Diabética", "Hipertensa", "Menor de Idade". Assim, fica fácil contar: "Quantas hipertensas temos hoje?".

Lógica Detalhada (CTEs)

  • SPLIT/UNNEST: Explode a string de riscos separados por ponto-e-vírgula em linhas individuais.
  • Limpeza: Remove espaços e registros vazios.

Principais Regras de Negócio

  • Granularidade: Garante 1 linha por risco por gestação para contagens exatas.

Código SQL Integral

CREATE OR REPLACE PROCEDURE `rj-sms-sandbox.sub_pav_us.proced_7_categorias_risco`()
BEGIN
CREATE OR REPLACE TABLE `rj-sms-sandbox.sub_pav_us._categorias_risco` AS
WITH riscos_separados AS (
  SELECT id_gestacao, TRIM(risco) AS categoria_risco FROM `rj-sms-sandbox.sub_pav_us._linha_tempo` left join UNNEST(SPLIT(categorias_risco, ';')) AS risco
  WHERE TRIM(risco) != ''
)
SELECT id_gestacao, categoria_risco FROM riscos_separados WHERE categoria_risco IS NOT NULL;
end;

8. Monitoramento de Sífilis (proced_8_sifilis.sql)

Objetivo

Acompanhamento rigoroso do tratamento de sífilis (Penicilina) e prevenção de sífilis congênita.

🛡️ Missão Zero Sífilis:

Sífilis na gravidez é muito perigoso, mas tem cura fácil. O problema é que o tratamento tem regras rígidas (3 doses, intervalo certo). Este robô é um auditor chato: ele conta os dias entre cada dose de Penicilina. Se a paciente atrasou um dia sequer, ele marca em vermelho: "Tratamento Falhou - Reiniciar". Ele não deixa passar nada.

Lógica Detalhada (CTEs)

  • Ciclos de Tratamento: Lógica de 21 dias para identificar novos ciclos de doses.
  • Doses 1, 2 e 3: Mapeia a data e o intervalo entre cada aplicação.
  • Monitoramento VDRL: Associa exames de diagnóstico e de cura ao tratamento.
  • Dados do Parceiro: Recupera o status do tratamento do parceiro.

Principais Regras de Negócio

  • Status Completo: Exige 3 doses em intervalos adequados.
  • Alerta de Falha: Identifica intervalos > 14 dias ou falta de acompanhamento laboratorial.

Código SQL Integral

CREATE OR REPLACE TABLE `rj-sms-sandbox.sub_pav_us._sifilis_gestantes` AS
WITH base_gestantes AS ( SELECT DISTINCT id_gestacao, id_paciente, cpf, SAFE.PARSE_DATE('%Y-%m-%d', SUBSTR(CAST(data_inicio AS STRING), 1, 10)) AS data_inicio_gestacao, COALESCE(SAFE.PARSE_DATE('%Y-%m-%d', SUBSTR(CAST(data_fim_efetiva AS STRING), 1, 10)), CURRENT_DATE('America/Sao_Paulo')) AS data_fim_gestacao FROM `rj-sms.projeto_gestacoes.gestacoes` WHERE fase_atual = 'Gestação' ),
diagnosticos_raw AS ( SELECT DISTINCT paciente.id_paciente, cond.id AS cid, SAFE.PARSE_DATE('%Y-%m-%d', SUBSTR(CAST(cond.data_diagnostico AS STRING), 1, 10)) AS data_diagnostico FROM `rj-sms.saude_historico_clinico.episodio_assistencial`, UNNEST(condicoes) AS cond WHERE (cond.id LIKE 'A51%' OR cond.id LIKE 'A53%' OR cond.id LIKE 'O981') AND cond.situacao = 'ATIVO' ),
dispensacoes_com_ciclo AS (
  SELECT id_gestacao, id_paciente, cpf, data_dispensacao, SUM(is_new_cycle) OVER (PARTITION BY id_gestacao ORDER BY data_dispensacao) AS ciclo_tratamento_id
  FROM (SELECT *, CASE WHEN data_dispensacao_anterior IS NULL OR DATE_DIFF(data_dispensacao, data_dispensacao_anterior, DAY) > 21 THEN 1 ELSE 0 END AS is_new_cycle FROM (SELECT *, LAG(data_dispensacao) OVER (PARTITION BY id_gestacao ORDER BY data_dispensacao) AS data_dispensacao_anterior FROM base_gestantes JOIN (SELECT DISTINCT consumo_paciente_cpf, SAFE.PARSE_DATE('%Y-%m-%d', SUBSTR(CAST(data_evento AS STRING), 1, 10)) AS data_dispensacao FROM `rj-sms.projeto_estoque.estoque_movimento` WHERE LOWER(material_descricao) LIKE '%benzilpenicilina%benzatina%') AS d ON cpf = d.consumo_paciente_cpf))
),
ciclos_de_tratamento AS (
  SELECT id_gestacao, id_paciente, cpf, ciclo_tratamento_id, MIN(CASE WHEN dose_num = 1 THEN data_dispensacao END) AS data_dispensacao_dose_1, COUNT(data_dispensacao) AS numero_doses_dispensadas FROM (SELECT *, ROW_NUMBER() OVER (PARTITION BY id_gestacao, ciclo_tratamento_id ORDER BY data_dispensacao) AS dose_num FROM dispensacoes_com_ciclo) GROUP BY 1,2,3,4
)
SELECT * FROM ciclos_de_tratamento;
